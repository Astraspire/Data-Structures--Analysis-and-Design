Pseudocode for CS300 Milestone One - Danny Fetter

DATA:
  COURSE is a structure with:
    courseId (text)
    name (text)
    coursePrerequisites (list of text)

  PENDING PREREQUISITE is a structure with:
    courseId (text)
    missingPrereqId (text)


  VARIABLES are:
    pendingPrerequisites is a list of PENDING PREREQUISITE
    newCourse is a COURSE
    courseToPrint is a COURSE
    courseCatalog is a list of COURSE
    /** The variable 'returnedIndexValue' is used by
      * the SearchCatalog() function 
      * changes the parameter returnedIndexValue 
      * If it returns: -1 = {SEARCH RETURNS NO RESULTS}, 
      * If it returns: -2 = {DEFAULT STATE}, or 
      * If it returns: i = {search result index}
      *
    returnedIndexValue is a number and starts at -2

  GLOBAL CONSTANTS:
  INPUT_FILE is the text "ABCU-CourseData.txt"


FUNCTIONS:
  FUNCTION ltrim(s)
    WHILE the first character of s is a space character
      remove the first character from s
    END WHILE
    RETURN s
  END FUNCTION

  FUNCTION Rtrim(s)
    WHILE the last character of s is a space character
      remove the last character from s
    END WHILE
    RETURN s
  END FUNCTION

  FUNCTION Ltrim(s)
    set s to rtrim(s)
    set s to ltrim(s)
    RETURN s
  END FUNCTION

  FUNCTION TrimString(inputString)
    set inputString to Rtrim(inputString)
    set inputString to Ltrim(inputString)
    RETURN inputString
  END FUNCTION

  FUNCTION AddCourseToCatalog(courseCatalog, newCourse)
    i is a number set to 0
    tempCourse is a COURSE

    FOR i from 0 to the size of courseCatalog minus 1
      IF newCourse.courseId is less than courseCatalog[i].courseId
        insert newCourse into courseCatalog at index i
        RETURN
      END IF
    END FOR

    add newCourse to the end of courseCatalog
  END FUNCTION

  FUNCTION PrerequisiteValidation(courseCatalog)
    i is a number set to 0
    j is a number set to 0
    searchOutput is a COURSE

    WHILE i is less than the size of pendingPrerequisites
      set searchOutput to output of SearchCatalog(courseCatalog, pendingPrerequisites[i].missingPrereqId)
      IF searchOutput is not empty
        FOR j from 0 to the size of courseCatalog minus 1
          IF courseCatalog[j].courseId equals pendingPrerequisites[i].courseId
            add pendingPrerequisites[i].missingPrereqId to courseCatalog[j].coursePrerequisites
          END IF
        END FOR
        remove pendingPrerequisites[i] from pendingPrerequisites
      ELSE
        increase i by 1
      END IF
    END WHILE
  END FUNCTION

  FUNCTION PrintCatalog(courseCatalog, returnedIndexValue default -1)
    ** -1 means no results so print all **
     *
    IF returnedIndexValue equals -1
      SET i to 0
      WHILE i is less than the size of courseCatalog
        set courseToPrint to courseCatalog[i]
        DISPLAY courseToPrint.courseId, ": ", courseToPrint.name
        IF the size of courseToPrint.coursePrerequisites is at least 1
          DISPLAY "Prerequisite Courses Needed:"
          SET j to 0
          WHILE j is less than the size of courseToPrint.coursePrerequisites
            DISPLAY j plus 1, ": ", courseToPrint.coursePrerequisites[j]
            increase j by 1
          END WHILE
        END IF
        increase i by 1
      END WHILE
    ELSE
      IF returnedIndexValue equals -2
        DISPLAY "Invalid index passed to print."
        RETURN
      END IF

      set courseToPrint to courseCatalog[returnedIndexValue]
      DISPLAY courseToPrint.courseId, ": ", courseToPrint.name
      IF the size of courseToPrint.coursePrerequisites is at least 1
        DISPLAY "Prerequisite Courses Needed:"
        SET j to 0
        WHILE j is less than the size of courseToPrint.coursePrerequisites
          DISPLAY j plus 1, ": ", courseToPrint.coursePrerequisites[j]
          increase j by 1
        END WHILE
      END IF
    END IF
  END FUNCTION

  FUNCTION SearchCatalog(courseCatalog, searchTerm)
    set searchTerm to trim(searchTerm)

    SET i to 0
    WHILE i is less than the size of courseCatalog
      IF searchTerm equals courseCatalog[i].courseId
        set returnedIndexValue to i
        RETURN
      END IF
      IF searchTerm equals courseCatalog[i].name
        set returnedIndexValue to i
        RETURN
      END IF
      increase i by 1
    END WHILE

    IF returnedIndexValue is greater than or equal to 0
      DISPLAY "  returned no results"
      set returnedIndexValue to -1
    END IF
  END FUNCTION

  FUNCTION SearchAndPrintCatalog(courseCatalog, searchTerm)
    call SearchCatalog(courseCatalog, searchTerm)

    IF returnedIndexValue equals -1
      DISPLAY "Type '1' to print all or 'Enter' to continue"
      read printChoice

      IF printChoice equals 1
        call PrintCatalog(courseCatalog)
      ELSE
        set returnedIndexValue to -2
      END IF

    ELSE
      IF returnedIndexValue equals -2
        DISPLAY "error, 'returnedIndexValue' is (-2)/default after search."
        RETURN
      ELSE
        call PrintCatalog(courseCatalog, returnedIndexValue)
        set returnedIndexValue to -2
      END IF
    END IF
  END FUNCTION

  FUNCTION ReadFile(fileName, courseCatalog)
    DISPLAY "Opening file " and fileName
    open file with name fileName for reading
    IF the file is not open
      DISPLAY "Could not open file"
      RETURN 1
    END IF

    DISPLAY "File open. Reading and storing data..."

    SET rawInput to empty text

    TRY
      WHILE there is another line to read from the file
        read the next line into rawInput

        SET current to empty text
        SET field to 0 
        * field indicators explaination:
        * 0 is for the courseId, 
        * 1 is for the name, 
        * 2+ are for prerequisites
        *
        CREATE a new COURSE named newCourse

        FOR EACH character ch in rawInput
          IF ch is a comma
            set current to TrimString(current)
            IF field equals 0
              set newCourse.courseId to current
            ELSE IF field equals 1
              set newCourse.name to current
            ELSE IF field is greater than 1
              searchOutput is a COURSE
              set searchOutput to output of SearchCatalog(courseCatalog, current)
              IF searchOutput is not empty
                add current to newCourse.coursePrerequisites
              ELSE
                CREATE a new PENDING PREREQUISITE named pendingPrereq
                set pendingPrereq.courseId to newCourse.courseId
                set pendingPrereq.missingPrereqId to current
                add pendingPrereq to pendingPrerequisites
              END IF
            set current to empty text
            increase field by 1
          ELSE
            add ch to the end of current
          END IF
        END FOR

        IF the length of current is at least 1
          set current to TrimString(current)
          IF field equals 1
            set newCourse.name to current
          ELSE IF field is greater than 1
            searchOutput is a COURSE
            set searchOutput to output of SearchCatalog(courseCatalog, current)
            IF searchOutput is not empty
              add current to newCourse.coursePrerequisites
            ELSE
              CREATE a new PENDING PREREQUISITE named pendingPrereq
              set pendingPrereq.courseId to newCourse.courseId
              set pendingPrereq.missingPrereqId to current
              add pendingPrereq to pendingPrerequisites
            END IF
          END IF
          set current to empty text
          increase field by 1
        END IF

        AddCourseToCatalog(courseCatalog, newCourse)
      END WHILE

      IF the file did not finish at the end marker
        DISPLAY "Input failure before reaching end of file."
      END IF

      * final validation pass to ensure all prerequisites are met *
      call PrerequisiteValidation(courseCatalog)
      DISPLAY the size of courseCatalog and " courses were added successfully!"
      DISPLAY "Closing file: " and fileName

      close the file

      DISPLAY "File: " and fileName
      DISPLAY "Import complete."

      RETURN 0
    CATCH exception
      DISPLAY "Error reading file ", csvPath
    END TRY
  END FUNCTION

  FUNCTION MainMenu()
    SET choice to 0

    DISPLAY "Press Enter to open the menu..."
    wait for the user to press Enter

    DO
      DISPLAY "MAIN MENU"
      DISPLAY "'1' to read the file."
      DISPLAY "'2' to print the entire course catalog."
      DISPLAY "'3' to search for and print a specific course."
      DISPLAY "'0' to exit."
      DISPLAY "User choice: "
      read choice
      clear any leftover input from that number entry

      IF choice equals 1
        call ReadFile(INPUT_FILE, courseCatalog)

      ELSE IF choice equals 2
        IF the size of courseCatalog is less than 1
          DISPLAY "No courses currently in the catalog"
        ELSE
          call PrintCatalog(courseCatalog)
        END IF

      ELSE IF choice equals 3
        IF the size of courseCatalog is less than 1
          DISPLAY "No courses currently in the catalog"
        ELSE
          SET searchTerm to empty text
          DISPLAY "  by course name or ID: "
          read a full line into searchTerm
          call SearchAndPrintCatalog(courseCatalog, searchTerm)
          DISPLAY "Press Enter to finish and return to main menu..."
          wait for the user to press Enter
        END IF

      ELSE IF choice equals 0
        (do nothing here)

      ELSE
        DISPLAY "Invalid entry!"
        DISPLAY "Press Enter to return to main menu..."
        wait for the user to press Enter
      END IF

    WHILE choice is not equal to 0

    DISPLAY "exiting..."
  END FUNCTION

MAIN PROGRAM
  call MainMenu()
  END PROGRAM
